# 跨平台内置音频播放实现计划

> **给智能体执行者：** 实施本计划时必须使用 `superpowers:subagent-driven-development`（推荐）或 `superpowers:executing-plans`，并按任务逐步勾选。步骤使用复选框 (`- [ ]`) 语法方便追踪。

**目标：** 实现第一版内置声音播放能力，覆盖 Web、Linux、Windows 和 Android 前台运行。

**架构：** React 负责 UI 和播放状态，所有播放操作通过 `PlayerPort` 接口完成。Web、Linux、Windows 使用 `public/audio` 和 WebAudio/HTMLAudio 适配器；Android 使用 Tauri 移动端桥接和 Kotlin `MediaPlayer`，但保持同一个前端契约。

**技术栈：** React 19、TypeScript、Vite、Vitest、Testing Library、Tauri 2、Rust、Android Kotlin、Android `MediaPlayer`。

  soundId: SoundId;
  isPlaying: boolean;
  volume: number;
}

export interface PlayerSnapshot {
  sounds: PlayingSoundState[];
}

export interface PlayerPort {
  play(sound: SoundDefinition, volume: number): Promise<void>;
  pause(soundId: SoundId): Promise<void>;
  setVolume(soundId: SoundId, volume: number): Promise<void>;
  stopAll(): Promise<void>;
  getState(): Promise<PlayerSnapshot>;
  destroy(): void;
}

export function normalizeVolume(volume: number): number {
  return Math.min(1, Math.max(0, volume));

```ts
import type { SoundDefinition, SoundId } from "../sounds/soundCatalog";
import type { PlayerPort, PlayerSnapshot } from "./PlayerPort";
import { normalizeVolume } from "./PlayerPort";

type AudioElementFactory = () => HTMLAudioElement;

export function createWebAudioPlayer(
  createAudio: AudioElementFactory = () => new Audio(),
): PlayerPort {
  const players = new Map<SoundId, HTMLAudioElement>();
  const volumes = new Map<SoundId, number>();

  function getAudio(sound: SoundDefinition): HTMLAudioElement {
    const existing = players.get(sound.id);
    if (existing) {
      return existing;
    }

    const audio = createAudio();
    audio.src = sound.sources[0].src;
    audio.loop = true;
    players.set(sound.id, audio);
    return audio;
  }

  return {
    async play(sound, volume) {
      const audio = getAudio(sound);
      const nextVolume = normalizeVolume(volume);
      volumes.set(sound.id, nextVolume);
      audio.volume = nextVolume;
      await audio.play();
    },

    async pause(soundId) {
      players.get(soundId)?.pause();
    },

    async setVolume(soundId, volume) {
      const nextVolume = normalizeVolume(volume);
      volumes.set(soundId, nextVolume);
      const audio = players.get(soundId);
      if (audio) {
        audio.volume = nextVolume;
      }
    },

    async stopAll() {
      for (const audio of players.values()) {
        audio.pause();
        audio.src = "";
        audio.load();
      }
      players.clear();
      volumes.clear();
    },

    async getState(): Promise<PlayerSnapshot> {
      return {
        sounds: Array.from(players.entries()).map(([soundId, audio]) => ({
          soundId,
          isPlaying: !audio.paused,
          volume: volumes.get(soundId) ?? audio.volume,
        })),
      };
    },

    destroy() {
      void this.stopAll();
    },
  };
}
```

import { invoke as tauriInvoke } from "@tauri-apps/api/core";
import type { SoundDefinition, SoundId } from "../sounds/soundCatalog";
import type { PlayerPort, PlayerSnapshot } from "./PlayerPort";
import { normalizeVolume } from "./PlayerPort";

type InvokeFn = <T>(cmd: string, args?: Record<string, unknown>) => Promise<T>;

export function createAndroidNativePlayer(
  invoke: InvokeFn = tauriInvoke,
): PlayerPort {
  return {
    async play(sound: SoundDefinition, volume: number) {
      await invoke("plugin:native-audio|play", {
        soundId: sound.id,
        resourceName: sound.androidResourceName,
        volume: normalizeVolume(volume),
      });
    },

    async pause(soundId: SoundId) {
      await invoke("plugin:native-audio|pause", { soundId });
    },

    async setVolume(soundId: SoundId, volume: number) {
      await invoke("plugin:native-audio|setVolume", {
        soundId,
        volume: normalizeVolume(volume),
      });
    },

    async stopAll() {
      await invoke("plugin:native-audio|stopAll");
    },

    async getState(): Promise<PlayerSnapshot> {
      return invoke<PlayerSnapshot>("plugin:native-audio|getState");
    },

    destroy() {
      void this.stopAll();
    },
  };
}

import type { PlayerPort } from "./PlayerPort";
import { createAndroidNativePlayer } from "./androidNativePlayer";
import { createWebAudioPlayer } from "./webAudioPlayer";

type RuntimePlatform = "web" | "linux" | "windows" | "android" | "unknown";

interface CreatePlayerOptions<TPlayer = PlayerPort> {
  detectPlatform?: () => Promise<RuntimePlatform>;
  createWebPlayer?: () => TPlayer;
  createAndroidPlayer?: () => TPlayer;
}

async function detectPlatform(): Promise<RuntimePlatform> {
  const maybeTauri = "__TAURI_INTERNALS__" in window;
  if (!maybeTauri) {
    return "web";
  }

  const { platform } = await import("@tauri-apps/plugin-os");
  const value = await platform();
  if (value === "android") {
    return "android";
  }
  if (value === "linux") {
    return "linux";
  }
  if (value === "windows") {
    return "windows";
  }
  return "unknown";
}

export async function createPlayer<TPlayer = PlayerPort>(
  options: CreatePlayerOptions<TPlayer> = {},
): Promise<TPlayer> {
  const platformDetector = options.detectPlatform ?? detectPlatform;
  const createWebPlayer =
    options.createWebPlayer ?? (() => createWebAudioPlayer() as TPlayer);
  const createAndroidPlayer =
    options.createAndroidPlayer ?? (() => createAndroidNativePlayer() as TPlayer);

  try {
    const runtimePlatform = await platformDetector();
    if (runtimePlatform === "android") {
      return createAndroidPlayer();
    }
  } catch {
    return createWebPlayer();
  }

  return createWebPlayer();
}
```

interface UseSoundMixerOptions {
  sounds: SoundDefinition[];
  player: PlayerPort;
}

export function useSoundMixer({ sounds, player }: UseSoundMixerOptions) {
  const soundById = useMemo(
    () => new Map(sounds.map((sound) => [sound.id, sound])),
    [sounds],
  );
  const [playingSoundIds, setPlayingSoundIds] = useState<Set<SoundId>>(
    () => new Set(),
  );
  const [volumes, setVolumes] = useState<Record<SoundId, number>>(() =>
    Object.fromEntries(sounds.map((sound) => [sound.id, 0.5])) as Record<
      SoundId,
      number
    >,
  );
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  const toggleSound = useCallback(
    async (soundId: SoundId) => {
      const sound = soundById.get(soundId);
      if (!sound) {
        return;
      }

      try {
        setErrorMessage(null);
        if (playingSoundIds.has(soundId)) {
          await player.pause(soundId);
          setPlayingSoundIds((current) => {
            const next = new Set(current);
            next.delete(soundId);
            return next;
          });
          return;
        }

        await player.play(sound, volumes[soundId]);
        setPlayingSoundIds((current) => new Set(current).add(soundId));
      } catch (error) {
        setErrorMessage(error instanceof Error ? error.message : "播放失败");
      }
    },
    [player, playingSoundIds, soundById, volumes],
  );

  const setSoundVolume = useCallback(
    async (soundId: SoundId, volume: number) => {
      const nextVolume = normalizeVolume(volume);
      setVolumes((current) => ({ ...current, [soundId]: nextVolume }));
      try {
        setErrorMessage(null);
        await player.setVolume(soundId, nextVolume);
      } catch (error) {
        setErrorMessage(error instanceof Error ? error.message : "音量调整失败");
      }
    },
    [player],
  );

  const stopAll = useCallback(async () => {
    try {
      setErrorMessage(null);
      await player.stopAll();
      setPlayingSoundIds(new Set());
    } catch (error) {
      setErrorMessage(error instanceof Error ? error.message : "停止播放失败");
    }
  }, [player]);

  return {
    playingSoundIds,
    volumes,
    errorMessage,
    toggleSound,
    setSoundVolume,
    stopAll,
  };
}
```
